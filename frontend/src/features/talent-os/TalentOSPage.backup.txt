
import { Link } from 'react-router'
import { useAuth } from '@/features/auth/useAuth'
import { GitHubRepositoriesSection } from '@/features/github/GitHubRepositoriesSection'
import './GitHubRepositoriesPage.css'

export function GitHubRepositoriesPage() {
  const { user, isLoading, error } = useAuth()

  if (isLoading) {
    return (
      <main className="talent-repositories">
        <p role="status">Đang kiểm tra phiên đăng nhập...</p>
      </main>
    )
  }

  return (
    <main className="talent-repositories">
      <div className="talent-repositories__top">
        <Link to="/talent-os" className="talent-repositories__back">
          ← Talent OS
        </Link>

        <span className="talent-repositories__label">
          GITHUB INTELLIGENCE
        </span>
      </div>

      <header className="talent-repositories__hero">
        <p className="talent-repositories__eyebrow">
          YOUR CODE. YOUR EVIDENCE.
        </p>

        <h1>GitHub Repositories</h1>

        <p>
          Đồng bộ repository, phân tích mã nguồn và khám phá
          năng lực lập trình từ dữ liệu GitHub thực tế.
        </p>
      </header>

      {error && (
        <p role="alert" className="talent-repositories__error">
          {error}
        </p>
      )}

      {!error && !user && (
        <section className="talent-repositories__empty">
          <h2>Chưa đăng nhập</h2>
          <p>
            Hãy đăng nhập bằng GitHub để xem repository của bạn.
          </p>
        </section>
      )}

      {user && (
        <GitHubRepositoriesSection isAuthenticated={true} />
      )}
    </main>
  )
}
